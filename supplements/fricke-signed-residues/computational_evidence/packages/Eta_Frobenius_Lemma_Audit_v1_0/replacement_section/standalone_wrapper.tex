\begin{theorem}[Frobenius on the eta quotient; statement unchanged]\label{thm:C}
Let $\ell\ne p$ be prime. At every supersingular point $x$ of $X_0(\ell)_{\Fb}$,
$\bigl(\Delta(\tau)/\Delta(\ell\tau)\bigr)^{(p^2-1)/24}(x)=\leg{\ell}{p}$.
For $\ell=N\in\mathcal N$ this reads $u_N(x)^{(p^2-1)/a_N}=\Lp N$.
\end{theorem}

\end{lemma}
